# One-off setup / cleanup script. Not invoked at runtime.
#
# Lock (Win+L) handling now lives ONLY in "LG TV TrayApp.ps1", which switches
# the TV to HDMI1 via its SessionSwitch handler. The scheduled task this script
# used to register did the opposite (-sethdmi1 as SYSTEM, with no reachability
# check) and would fight the tray app on every lock, so it is no longer created
# and any previously registered copy is removed below.
#
# This script also no longer embeds its own copy of the toggle logic. That was a
# third duplicate that silently overwrote "LG TV Toggle.ps1"; the canonical
# version is that file itself.

# --- YOUR SPECIFIC CONFIGURATION PATH ---
$InstallDir = "C:\Users\joshi\OneDrive - Monarch Innovation Private Limited\Documents\Anitgravity\LG TV"
if (-not (Test-Path $InstallDir)) { New-Item -ItemType Directory -Path $InstallDir -Force | Out-Null }

$PsScriptPath = "$InstallDir\LG TV Toggle.ps1"
$VbsScriptPath = "$InstallDir\LG TV Toggle.vbs"

# --- 1. SANITY-CHECK THE TOGGLE SCRIPT ---
if (-not (Test-Path $PsScriptPath)) {
    Write-Warning "Missing $PsScriptPath - restore it from the repo before continuing."
    return
}

# --- 2. (RE)WRITE THE INVISIBLE VBS LAUNCHER ---
# Purely derived from the path above, so it is safe to regenerate.
$VbsCode = @"
Set WshShell = CreateObject("WScript.Shell")
WshShell.Run "powershell.exe -WindowStyle Hidden -ExecutionPolicy Bypass -File ""$PsScriptPath""", 0, False
"@
Set-Content -Path $VbsScriptPath -Value $VbsCode -Force

# --- 3. REMOVE THE OBSOLETE ON-LOCK TASK ---
$TaskName = "LGTV_Switch_On_Lock"
if (Get-ScheduledTask -TaskName $TaskName -ErrorAction SilentlyContinue) {
    Unregister-ScheduledTask -TaskName $TaskName -Confirm:$false
    Write-Host "Removed obsolete scheduled task '$TaskName'." -ForegroundColor Yellow
} else {
    Write-Host "No '$TaskName' task registered - nothing to remove." -ForegroundColor DarkGray
}

Write-Host ""
Write-Host "=============================================" -ForegroundColor Green
Write-Host "  Done. Lock handling is owned by the tray app" -ForegroundColor Green
Write-Host "=============================================" -ForegroundColor Green
